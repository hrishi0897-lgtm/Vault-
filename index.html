<script>
let backendUrl = "http://localhost:3000"; // Change to your Render/Railway live backend URL after deployment
let vaultFiles = JSON.parse(localStorage.getItem('vault_files')) || [];

function openUploadModal() { document.getElementById('uploadModal').classList.remove('hidden'); }
function closeUploadModal() { document.getElementById('uploadModal').classList.add('hidden'); }
function openSettingsModal() {
document.getElementById('botTokenInput').value = localStorage.getItem('botToken') || '';
document.getElementById('chatIdInput').value = localStorage.getItem('chatId') || '';
document.getElementById('settingsModal').classList.remove('hidden');
}
function closeSettingsModal() { document.getElementById('settingsModal').classList.add('hidden'); }

function saveSettings() {
localStorage.setItem('botToken', document.getElementById('botTokenInput').value);
localStorage.setItem('chatId', document.getElementById('chatIdInput').value);
closeSettingsModal();
alert('Settings saved!');
}

async function startUpload() {
const fileInput = document.getElementById('fileInput');
if (fileInput.files.length === 0) return alert('Select a file first!');

const botToken = localStorage.getItem('botToken');
const chatId = localStorage.getItem('chatId');
if (!botToken || !chatId) return alert('Please configure your bot token and chat ID in settings first!');

const formData = new FormData();
formData.append('file', fileInput.files[0]);
formData.append('botToken', botToken);
formData.append('chatId', chatId);

alert('Uploading and chunking file to Telegram via backend...');
try {
const res = await fetch(`${backendUrl}/api/upload`, { method: 'POST', body: formData });
const data = await res.json();
if(data.success) {
vaultFiles.unshift(data);
localStorage.setItem('vault_files', JSON.stringify(vaultFiles));
closeUploadModal();
renderFiles();
alert('File uploaded successfully!');
} else {
alert('Upload failed: ' + data.error);
}
} catch (err) {
alert('Error connecting to backend server.');
}
}

function renderFiles() {
const container = document.getElementById('filesContainer');
container.innerHTML = vaultFiles.map((file, index) => `
<div class="glass-panel p-4 rounded-xl space-y-3">
<p class="text-sm font-semibold truncate">${file.fileName}</p>
<p class="text-xs text-slate-400">Parts: ${file.parts}</p>
<button onclick="downloadFile(${index})" class="w-full py-2 bg-cyan-500/20 text-cyan-400 rounded-lg text-xs font-semibold">Reassemble & Download</button>
</div>
`).join('');
}

async function downloadFile(index) {
const file = vaultFiles[index];
const botToken = localStorage.getItem('botToken');
alert('Reassembling chunks from Telegram server...');

try {
const res = await fetch(`${backendUrl}/api/download`, {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify({ botToken, telegramMessages: file.telegramMessages, fileName: file.fileName })
});
const blob = await res.blob();
const url = window.URL.createObjectURL(blob);
const a = document.createElement('a');
a.href = url;
a.download = file.fileName;
a.click();
} catch(e) {
alert('Download failed');
}
}
renderFiles();
</script>
